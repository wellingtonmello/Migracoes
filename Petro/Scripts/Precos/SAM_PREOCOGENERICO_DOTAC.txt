begin
 --LIMPEZA DE TABELAS
DELETE FROM SAM_PRECOPRESTADOR_DOTAC;

UPDATE TMP_MIG_SAM_PRECOPRESTADOR_DOTAC
  SET ERRO = NULL,
      HANDLEPRESTADOR= NULL,
      HANDLEEVENTO = NULL;
      
MERGE INTO TMP_MIG_SAM_PRECOPRESTADOR_DOTAC TMP
USING SAM_PRESTADOR P ON (TMP.PRESTADOR = P.PRESTADOR)
WHEN MATCHED THEN
  UPDATE SET TMP.HANDLEPRESTADOR = P.HANDLE;
  
MERGE INTO TMP_MIG_SAM_PRECOPRESTADOR_DOTAC TMP
USING SAM_TGE E ON (TMP.EVENTO = E.ESTRUTURANUMERICA AND TMP.ARQUIVO = 'dtx.csv' and E.MASCARATGE = 1) --TUSS
WHEN MATCHED THEN
  UPDATE SET TMP.HANDLEEVENTO = E.HANDLE
  WHERE TMP.ARQUIVO = 'dtx.csv'
    AND HANDLEEVENTO IS NULL ; 

MERGE INTO TMP_MIG_SAM_PRECOPRESTADOR_DOTAC TMP
USING SAM_TGE E ON (TMP.EVENTO = E.ESTRUTURANUMERICA AND TMP.ARQUIVO = 'dtx.csv' and E.MASCARATGE <> 1 ) --NÃO TUSS
WHEN MATCHED THEN
  UPDATE SET TMP.HANDLEEVENTO = E.HANDLE
  WHERE TMP.ARQUIVO = 'dtx.csv'
    AND HANDLEEVENTO IS NULL;
  
  
UPDATE TMP_MIG_SAM_PRECOPRESTADOR_DOTAC SET ERRO = ERRO || 'EVENTO NAO ENCONTRADO' WHERE HANDLEEVENTO IS NULL AND ARQUIVO = 'dtx.csv';


  
MERGE INTO TMP_MIG_SAM_PRECOPRESTADOR_DOTAC TMP
USING SAM_TGE E ON (TMP.EVENTO = E.ESTRUTURANUMERICA AND TMP.ARQUIVO = 'pro.csv' and E.MASCARATGE <> 1) --NAO TUSS
WHEN MATCHED THEN
  UPDATE SET TMP.HANDLEEVENTO = E.HANDLE
  WHERE TMP.ARQUIVO = 'pro.csv'
    AND HANDLEEVENTO IS NULL;

MERGE INTO TMP_MIG_SAM_PRECOPRESTADOR_DOTAC TMP
USING SAM_TGE E ON (TMP.EVENTO = E.ESTRUTURANUMERICA AND TMP.ARQUIVO = 'pro.csv' and E.MASCARATGE = 1) --TUSS
WHEN MATCHED THEN
  UPDATE SET TMP.HANDLEEVENTO = E.HANDLE
  WHERE TMP.ARQUIVO = 'pro.csv'
    AND HANDLEEVENTO IS NULL ; 
    
    
UPDATE TMP_MIG_SAM_PRECOPRESTADOR_DOTAC SET ERRO = ERRO || 'EVENTO NAO ENCONTRADO' WHERE HANDLEEVENTO IS NULL AND ARQUIVO = 'pro.csv'; 

MERGE INTO TMP_MIG_SAM_PRECOPRESTADOR_DOTAC TMP
USING SAM_TGE E ON (TMP.EVENTO = E.ESTRUTURANUMERICA AND TMP.ARQUIVO = 'hon.csv' and E.MASCARATGE = 1) --TUSS
WHEN MATCHED THEN
  UPDATE SET TMP.HANDLEEVENTO = E.HANDLE
  WHERE TMP.ARQUIVO = 'hon.csv'
    AND HANDLEEVENTO IS NULL;  

MERGE INTO TMP_MIG_SAM_PRECOPRESTADOR_DOTAC TMP
USING SAM_TGE E ON (TMP.EVENTO = E.ESTRUTURANUMERICA AND TMP.ARQUIVO = 'hon.csv' and E.MASCARATGE <> 1) --CBHPM
WHEN MATCHED THEN
  UPDATE SET TMP.HANDLEEVENTO = E.HANDLE
  WHERE TMP.ARQUIVO = 'hon.csv'
    AND HANDLEEVENTO IS NULL;      
    
    
UPDATE TMP_MIG_SAM_PRECOPRESTADOR_DOTAC SET ERRO = ERRO || 'EVENTO NAO ENCONTRADO' WHERE HANDLEEVENTO IS NULL AND ARQUIVO = 'hon.csv'; 


MERGE INTO TMP_MIG_SAM_PRECOPRESTADOR_DOTAC TMP
USING SAM_TGE E ON (TMP.EVENTO = E.ESTRUTURANUMERICA AND TMP.ARQUIVO = 'exc.csv' and E.MASCARATGE = 1) --TUSS
WHEN MATCHED THEN
  UPDATE SET TMP.HANDLEEVENTO = E.HANDLE
  WHERE TMP.ARQUIVO = 'exc.csv'
    AND HANDLEEVENTO IS NULL;  

MERGE INTO TMP_MIG_SAM_PRECOPRESTADOR_DOTAC TMP
USING SAM_TGE E ON (TMP.EVENTO = E.ESTRUTURANUMERICA AND TMP.ARQUIVO = 'exc.csv' and E.MASCARATGE <> 1) --NAO TUSS
WHEN MATCHED THEN
  UPDATE SET TMP.HANDLEEVENTO = E.HANDLE
  WHERE TMP.ARQUIVO = 'exc.csv'
    AND HANDLEEVENTO IS NULL;

UPDATE TMP_MIG_SAM_PRECOPRESTADOR_DOTAC SET ERRO = ERRO || 'EVENTO NAO ENCONTRADO' WHERE HANDLEEVENTO IS NULL AND ARQUIVO = 'exc.csv';

--INSERIR HONORARIOS

UPDATE TMP_MIG_SAM_PRECOPRESTADOR_DOTAC D1 SET ERRO = ERRO || 'EXISTE EXCEÇÃO PARA O REGISTRO'
 WHERE D1.ARQUIVO <> 'exc.csv'
   AND EXISTS 
        (SELECT 1
          FROM TMP_MIG_SAM_PRECOPRESTADOR_DOTAC D2
         WHERE (
                            (D2.DATAINICIAL BETWEEN D1.DATAINICIAL and D1.DATAFINAL) or
                            (D2.DATAFINAL BETWEEN D1.DATAINICIAL and D1.DATAFINAL) or
                            (D1.DATAFINAL IS NULL AND D2.DATAFINAL IS NULL) OR
                            (D1.DATAFINAL IS NULL AND (D2.DATAFINAL >= D1.DATAINICIAL)) OR
                            (D2.DATAINICIAL <= D1.DATAINICIAL AND D2.DATAFINAL >= D1.DATAINICIAL)
               )
           AND D2.PRESTADOR = D1.PRESTADOR
           AND D2.ARQUIVO = 'exc.csv'
           AND D2.EVENTO = D1.EVENTO
           AND D2.ROWID <> D1.ROWID
        ); 
        
UPDATE TMP_MIG_SAM_PRECOPRESTADOR_DOTAC D1 SET D1.ERRO = D1.ERRO || 'VIGENCIA EXCECAO DUPLICADA'
 WHERE D1.ARQUIVO = 'exc.csv'
    AND EXISTS
       (SELECT 1
          FROM TMP_MIG_SAM_PRECOPRESTADOR_DOTAC D2
         WHERE (
                            (D2.DATAINICIAL BETWEEN D1.DATAINICIAL and D1.DATAFINAL) or
                            (D2.DATAFINAL BETWEEN D1.DATAINICIAL and D1.DATAFINAL) or
                            (D1.DATAFINAL IS NULL AND D2.DATAFINAL IS NULL) OR
                            (D1.DATAFINAL IS NULL AND (D2.DATAFINAL >= D1.DATAINICIAL)) OR
                            (D2.DATAINICIAL <= D1.DATAINICIAL AND D2.DATAFINAL >= D1.DATAINICIAL)
               )
           AND D2.PRESTADOR = D1.PRESTADOR
           AND D2.EVENTO = D1.EVENTO
           AND D2.ARQUIVO = 'exc.csv'
           AND D2.ROWID <> D1.ROWID
        );

        
UPDATE TMP_MIG_SAM_PRECOPRESTADOR_DOTAC D1 SET D1.ERRO = D1.ERRO || 'VIGENCIA NAO EXCECAO DUPLICADA '
 WHERE  D1.ARQUIVO <> 'exc.csv'
   AND EXISTS
       (SELECT 1
          FROM TMP_MIG_SAM_PRECOPRESTADOR_DOTAC D2
         WHERE (
                            (D2.DATAINICIAL BETWEEN D1.DATAINICIAL and D1.DATAFINAL) or
                            (D2.DATAFINAL BETWEEN D1.DATAINICIAL and D1.DATAFINAL) or
                            (D1.DATAFINAL IS NULL AND D2.DATAFINAL IS NULL) OR
                            (D1.DATAFINAL IS NULL AND (D2.DATAFINAL >= D1.DATAINICIAL)) OR
                            (D2.DATAINICIAL <= D1.DATAINICIAL AND D2.DATAFINAL >= D1.DATAINICIAL)
               )
           AND D2.PRESTADOR = D1.PRESTADOR
           AND D2.EVENTO = D1.EVENTO
           AND D2.ARQUIVO <> 'exc.csv'
           AND D2.ROWID <> D1.ROWID
        );

INSERT INTO SAM_PRECOPRESTADOR_DOTAC
(HANDLE,
Z_GRUPO,
PERCENTUALPGTOFILME,
DATAINICIAL,
DATAFINAL,
EVENTO,
FATORFILME,
TABELAUS,
TABELAFILME,
PRESTADOR,
CLASSEASSOCIADO,
PERCENTUALPGTOUS,
TABELACUSTOOPERAC,
PERCENTUALPGTOCUSTO,
CONVENIO,
PORTEANESTESICO,
PORTESALA,
QTDUSHONORARIO,
QTDUSCUSTOOPERACIONAL,
CBOS,
CBOSPESQUISA,
TABNEGOCIACAO,
DEANOS,
DEDIAS,
ATEANOS,
ATEDIAS)
SELECT
SEQ_SAM0006.NEXTVAL HANDLE,
Z_GRUPO,
PERCENTUALPGTOFILME,
DATAINICIAL,
DATAFINAL,
HANDLEEVENTO,
FATORFILME,
TABELAUS,
TABELAFILME,
HANDLEPRESTADOR,
CLASSEASSOCIADO,
PERCENTUALPGTOUS,
TABELACUSTOOPERAC,
PERCENTUALPGTOCUSTO,
CONVENIO,
PORTEANESTESICO,
PORTESALA,
QTDUSHONORARIO,
QTDUSCUSTOOPERACIONAL,
CBOS,
CBOSPESQUISA,
TABNEGOCIACAO,
DEANOS,
DEDIAS,
ATEANOS,
ATEDIAS
FROM TMP_MIG_SAM_PRECOPRESTADOR_DOTAC 
WHERE ERRO IS NULL;
commit;
end;
