SELECT LPAD(TO_NUMBER(C.CONTRATO),2,0)||LPAD(B.MATRICULAFUNCIONAL,8,0)||LPAD(TO_NUMBER(B.CODIGODEPENDENTE),2,0)CARTEIRINHA,
       B.NOME,
       C.CONTRATO,
       FAM.FAMILIA,
       B.MATRICULAFUNCIONAL,
       CRITICABENEF.CODIGO CODIGOERROBENEF,
       CRITICABENEF.DESCRICAO DESCRICAOERROBENEF,
       CRITICAFAM.CODIGO CODIGOERROFAMILIA,
       CRITICAFAM.DESCRICAO DESCRICAOERROFAMILIA
 FROM SAM_ROTINAIMP_BENEF B
 JOIN SAM_CONTRATO C ON (C.HANDLE = B.CONTRATO)
 LEFT JOIN SAM_ROTINAIMP_BENEF_ERRO BERRO ON (BERRO.ROTINAIMPBENEF = B.HANDLE)
 left JOIN SAM_ROTINAIMP_FAM FAM ON (B.IMPFAM = FAM.HANDLE)
 LEFT JOIN SAM_ROTINAIMP_FAM_ERRO FERRO ON (FERRO.ROTINAIMPFAM = FAM.HANDLE)
 LEFT JOIN SIS_CRITICAIMPORTACAO CRITICABENEF ON (CRITICABENEF.HANDLE = BERRO.CODIGOERRO)
 LEFT JOIN SIS_CRITICAIMPORTACAO CRITICAFAM ON (CRITICAFAM.HANDLE = FERRO.CODIGOERRO)
WHERE (FERRO.HANDLE IS NOT NULL OR BERRO.HANDLE IS NOT NULL)
ORDER BY CRITICABENEF.CODIGO, CRITICAFAM.CODIGO
