SELECT md.comments, COUNT(*)
FROM AZU200_BS_HML.MSG_PACKAGES mp
INNER JOIN AZU200_BS_HML.AUT_ACCOUNTS aa ON aa.HANDLE = mp.ACCOUNT  
INNER JOIN AZU200_BS_HML.MSG_DATA md ON md."PACKAGE" = mp.HANDLE 
WHERE mp.HANDLE between 5840 AND 8000 -- LOTEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEE
  AND aa.NAME LIKE 'IMPORTACAO_HOMOLOGACAO%'
  AND md.COMMENTS IS NOT null  
GROUP BY md.comments
ORDER BY 2 DESC
